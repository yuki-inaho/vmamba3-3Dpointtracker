cmake_minimum_required(VERSION 3.20)
project(mamba3_refiner_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted official ONNX Runtime CPU SDK")
if(NOT EXISTS "${ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set -DONNXRUNTIME_ROOT=/path/to/onnxruntime-linux-x64-1.30.0")
endif()
find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(refiner_runner refiner_runner.cpp)
target_include_directories(refiner_runner SYSTEM PRIVATE "${ONNXRUNTIME_ROOT}/include")
target_link_libraries(refiner_runner PRIVATE "${ORT_LIBRARY}")
if(MSVC)
  target_compile_options(refiner_runner PRIVATE /W4 /WX)
else()
  target_compile_options(refiner_runner PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
# Installed executable is relocatable with its official runtime alongside it.
set_target_properties(refiner_runner PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS refiner_runner RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(DIRECTORY "${ONNXRUNTIME_ROOT}/lib/" DESTINATION lib
          FILES_MATCHING PATTERN "libonnxruntime.so*")
  install(FILES "${ONNXRUNTIME_ROOT}/LICENSE" "${ONNXRUNTIME_ROOT}/ThirdPartyNotices.txt"
          DESTINATION share/licenses/onnxruntime)
endif()
